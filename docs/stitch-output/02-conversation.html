import serial
import time

def monitor_vreg_thermal(port: str = "/dev/ttyUSB0", baud: int = 115200):
    # Inicia handshake com a bancada de testes de bancada
    ser = serial.Serial(port, baud, timeout=1.0)
    print(f"[*] Conectado ao nó de instrumentação em {port}")

    while True:
        ser.write(b"GET_THERMAL_U3\n")
        raw = ser.readline().decode().strip()
        if raw.startswith("TEMP:"):
            celsius = float(raw.split(":")[1])
            if celsius > 85.0:
                print(f"[ALERTA CRÍTICO] Junção U3 em {celsius}°C - Throttling necessário!")
        time.sleep(0.2)
