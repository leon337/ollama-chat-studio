Abaixo está a implementação com anotações de alinhamento #[repr(align(64))] e canal lock-free baseado em ring buffer de anel estático:

src/scheduler.rs (Rust 2024) 
use crossbeam_channel::{bounded, Receiver, Sender};
use core_affinity::CoreId;

#[repr(align(64))]
pub struct HotTaskBuffer<T, const N: usize> {
    tx: Sender<T>,
    rx: Receiver<T>,
}

impl<T, const N: usize> HotTaskBuffer<T, N> {
    pub fn new_pinned(core_id: CoreId) -> Self {
        core_affinity::set_for_current(core_id);
        let (tx, rx) = bounded(N);
        Self { tx, rx }
    }
}
